```python
import hashlib
import json


def selection_hash(raw_sha256: str, selection: dict[str, object]) -> str:
    payload = {"raw_sha256": raw_sha256, "selection": selection}
    encoded = json.dumps(payload, sort_keys=True, separators=(",", ":"),
                         ensure_ascii=False, allow_nan=False).encode("utf-8")
    return hashlib.sha256(encoded).hexdigest()
```
